#### Captura de `WordBoundary` no `EdgeTTSVoiceSynthesizer`
O `communicate.stream()` do Edge-TTS produz chunks de dois tipos: `"audio"` (MP3 binário) e `"WordBoundary"` (marcações temporais com `offset` e `duration` em unidades de 100ns).
A conversão para segundos `float` divide por $10^7$ (`10_000_000.0`).

```python
class EdgeTTSVoiceSynthesizer(BaseVoiceSynthesizer):
    def __init__(self, voice: str | None = None, rate: str | None = None):
        settings = get_settings()
        self.voice = voice or settings.edge_tts_voice
        self.rate = rate or settings.edge_tts_rate
        self.last_word_timestamps: list[dict] | None = None

    def synthesize(
        self,
        text: str,
        output_path: Path,
        speaker_wav: Path | None = None,
    ) -> tuple[Path, float]:
        output_path.parent.mkdir(parents=True, exist_ok=True)
        temp_mp3 = output_path.with_suffix(".edge.temp.mp3")
        word_timestamps: list[dict] = []

        try:
            import edge_tts

            async def _run_stream():
                communicate = edge_tts.Communicate(
                    text, self.voice, rate=self.rate, boundary="WordBoundary"
                )
                with open(temp_mp3, "wb") as f_out:
                    async for chunk in communicate.stream():
                        if chunk["type"] == "audio":
                            f_out.write(chunk["data"])
                        elif chunk["type"] == "WordBoundary":
                            start_sec = chunk["offset"] / 10_000_000.0
                            dur_sec = chunk["duration"] / 10_000_000.0
                            end_sec = start_sec + dur_sec
                            word_timestamps.append({
                                "word": chunk["text"],
                                "start": round(start_sec, 4),
                                "end": round(end_sec, 4),
                                "duration": round(dur_sec, 4),
                            })

            asyncio.run(_run_stream())

            if not shutil.which("ffmpeg"):
                raise RuntimeError("FFmpeg é obrigatório para converter a síntese Edge-TTS MP3 para WAV.")

            if not temp_mp3.exists() or temp_mp3.stat().st_size == 0:
                raise RuntimeError("Falha na síntese Edge-TTS: arquivo MP3 intermediário não foi gerado.")

            # Normalização acústica comprovada com desvio nulo (diff <= 0.05s)
            subprocess.run(
                [
                    "ffmpeg", "-y", "-i", str(temp_mp3),
                    "-af", "loudnorm=I=-16:TP=-1.5:LRA=11",
                    "-ac", "1", "-ar", "24000", str(output_path),
                ],
                check=True,
                capture_output=True,
                timeout=60,
            )

            duration = probe_duration(output_path)
            if duration <= 0.0:
                raise RuntimeError(f"FFprobe retornou duração inválida ({duration}s) para {output_path}")

            # Persistência de estado e sidecar
            self.last_word_timestamps = word_timestamps
            sidecar_path = output_path.with_suffix(".words.json")
            sidecar_path.write_text(
                json.dumps(word_timestamps, indent=2, ensure_ascii=False),
                encoding="utf-8",
            )
            return output_path, duration
        finally:
            if temp_mp3.exists():
                temp_mp3.unlink(missing_ok=True)
```

#### Alinhador Resiliente de Tokens (`align_words_with_timestamps`)
Pareia `script_text.split()` com os eventos `WordBoundary` do Edge-TTS, tratando pontuações, hífens, numerais compostos e contrações sem disparar exceções. Aplica clamp em $[0.0, \text{total\_duration}]$.

```python
def align_words_with_timestamps(
    script_words: list[str],
    word_timestamps: list[dict] | None,
    total_duration: float,
) -> list[dict]:
    """Pareia tokens do roteiro com timestamps do Edge-TTS de forma resiliente.

    Trata pontuações, numerais compostos e contrações sem disparar exceções.
    Aplica clamp em [0.0, total_duration] e interpolação proporcional em caso de gaps.
    """
    if not script_words:
        return []

    if not word_timestamps:
        # Fallback proporcional caso a lista de timestamps seja vazia
        char_counts = [len(w) for w in script_words]
        total_chars = max(sum(char_counts), 1)
        cur = 0.0
        result = []
        for w, c in zip(script_words, char_counts):
            dur = (c / total_chars) * total_duration
            result.append({
                "word": w,
                "start": round(cur, 3),
                "end": round(cur + dur, 3),
                "duration": round(dur, 3),
            })
            cur += dur
        return result

    aligned: list[dict] = []
    wb_idx = 0
    num_wb = len(word_timestamps)
    last_end = 0.0

    for word in script_words:
        clean_word = re.sub(r"[^\w]", "", word.lower())

        if wb_idx < num_wb:
            wb = word_timestamps[wb_idx]
            clean_wb = re.sub(r"[^\w]", "", wb["word"].lower())

            # Correspondência direta ou substring mútua
            if clean_word == clean_wb or clean_word in clean_wb or clean_wb in clean_word:
                start = max(wb["start"], last_end)
                end = min(max(wb["end"], start + 0.05), total_duration)
                aligned.append({
                    "word": word,
                    "start": round(start, 3),
                    "end": round(end, 3),
                    "duration": round(end - start, 3),
                })
                last_end = end
                wb_idx += 1
                continue

        # Interpolação defensiva se token pulado pelo TTS
        start = last_end
        end = min(start + 0.25, total_duration)
        aligned.append({
            "word": word,
            "start": round(start, 3),
            "end": round(end, 3),
            "duration": round(end - start, 3),
        })
        last_end = end

    return aligned
```

#### Gerador de Eventos ASS com Active Word Highlight & Contiguidade Centesimal (D-03, D-04, D-05)
```python
def _format_cs(cs: int) -> str:
    """Converte centésimos de segundo inteiros para timestamp ASS (H:MM:SS.cs)."""
    hours = cs // 360000
    mins = (cs % 360000) // 6000
    secs = (cs % 6000) // 100
    centis = cs % 100
    return f"{hours}:{mins:02d}:{secs:02d}.{centis:02d}"


def generate_word_karaoke_dialogue_events(
    cue_words: list[dict],
    yellow_ass_color: str,
    white_ass_color: str,
) -> list[str]:
    """Gera eventos Dialogue sequenciais por palavra ativa para um cue ZUNTO!.

    Regras:
    - As palavras já faladas (0..i-1) voltam a branco puro ({\c&H00FFFFFF&}).
    - Apenas a palavra ativa i recebe amarelo ZUNTO! ({\c&H0015CCFA&}).
    - Palavras futuras (i+1..K-1) permanecem em branco puro.
    - Piso visual: mínimo 50ms (5 cs) para 24 FPS (D-04).
    - Pausas curtas (<= 120ms): destaque amarelo é estendido até o início da próxima palavra.
    - Pausas longas (> 120ms): término do destaque e emissão de evento neutro contíguo em branco.
    - Contiguidade estrita em centésimos inteiros: event[k].end_cs == event[k+1].start_cs (D-03, D-05).
    """
    if not cue_words:
        return []

    events: list[str] = []
    k_words = len(cue_words)
    raw_text_words = [sanitize_ass_text(w["word"]) for w in cue_words]

    for i in range(k_words):
        w_curr = cue_words[i]
        start_cs = max(0, int(round(w_curr["start"] * 100)))
        end_cs = max(0, int(round(w_curr["end"] * 100)))

        # D-04: Piso mínimo de 50ms
        if end_cs - start_cs < 5:
            end_cs = start_cs + 5

        # D-05: Avaliação de pausa até a próxima palavra do mesmo cue
        if i < k_words - 1:
            next_start_cs = max(0, int(round(cue_words[i + 1]["start"] * 100)))
            gap_cs = next_start_cs - end_cs

            if 0 < gap_cs <= 12:  # <= 120ms: absorve a pausa
                end_cs = next_start_cs

        # Constrói a linha com apenas a palavra ativa em amarelo e as demais em branco
        formatted_words = []
        for idx, text in enumerate(raw_text_words):
            if idx == i:
                formatted_words.append(f"{{\\c{yellow_ass_color}}}{text}{{\\c{white_ass_color}}}")
            else:
                formatted_words.append(text)

        line_text = " ".join(formatted_words)
        events.append(f"Dialogue: 0,{_format_cs(start_cs)},{_format_cs(end_cs)},Default,,0,0,0,,{line_text}")

        # Se houver pausa longa (> 120ms) intra-bloco, emite evento neutro em branco para manter a caixa
        if i < k_words - 1:
            next_start_cs = max(0, int(round(cue_words[i + 1]["start"] * 100)))
            if next_start_cs - end_cs > 0:
                neutral_text = " ".join(raw_text_words)
                events.append(f"Dialogue: 0,{_format_cs(end_cs)},{_format_cs(next_start_cs)},Default,,0,0,0,,{neutral_text}")

    return events
```

## Cross-Cutting Patterns & Shared Conventions

### 1. Representação de Cores ASS (BGR vs RGB)
No ASS, a ordem de cores no canal hexadecimal é BGR (Blue-Green-Red):
- `IMPACT_YELLOW = "#FACC15"` (RGB: R=250, G=204, B=21) $\longrightarrow$ ASS: `&H0015CCFA&`
- `PURE_WHITE = "#FFFFFF"` $\longrightarrow$ ASS: `&H00FFFFFF&`
- `DEEP_BLACK = "#0A0A0A"` $\longrightarrow$ ASS: `&H000A0A0A&`
Sempre utilizar a função canônica [hex_to_ass_color](file:///c:/Users/Windows%20Lite%20BR/Documents/GitHub/viralpanel/src/viralpanel/branding.py#L70-L103) para evitar divergências.

### 2. Aritmética Temporal Inteira em Centésimos de Segundo
Para eliminar discrepâncias de ponto flutuante e evitar piscadas da caixa opaca (`BorderStyle=3`), toda a lógica de contiguidade sequencial opera em inteiros:
$$\text{cs} = \max(0, \text{int}(\text{round}(t \times 100)))$$
A invariante obrigatória dentro de cada cue é:
$$\text{event}[k+1].\text{start\_cs} = \text{event}[k].\text{end\_cs}$$

### 3. Piso de Visibilidade Sub-frame (24 FPS)
A taxa padrão de renderização do projeto é 24 FPS (~41.67 ms por quadro). Nenhuma palavra ativa pode ter duração inferior a 50 ms ($5\text{ cs}$), garantindo captura em ao menos um quadro de vídeo.

### 4. Gestão de Pausas e Estabilidade da Caixa Opaca
- **Pausas intra-bloco curtas ($\le 120\text{ms}$ / 12 cs):** Absorvidas estendendo o destaque da palavra corrente até o início da próxima.
- **Pausas intra-bloco longas ($> 120\text{ms}$):** Emissão de evento neutro contíguo com todas as palavras em branco puro para manter a caixa preta na tela.
- **Pausas inter-blocos (troca de frase):** Início do novo bloco sincronizado rigorosamente com a primeira palavra pronunciada.
